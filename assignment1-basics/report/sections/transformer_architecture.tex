\problem{linear}{Implementing the linear module}{1}

\deliverable{Implement a \texttt{Linear} class that inherits from \texttt{torch.nn.Module} and performs a linear transformation.}

% Implementation summary:
The bias-free weight has shape $(d_{\mathrm{out}}, d_{\mathrm{in}})$; an Einstein summation projects only the final input dimension.
\par\noindent
\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={Linear projection}]
def forward(self, x: Tensor) -> Tensor:
    return einsum(x, self.weight, "... d_in, d_out d_in -> ... d_out")
\end{lstlisting}
\end{minipage}\par

\problem{embedding}{Implement the embedding module}{1}

\deliverable{Implement an \texttt{Embedding} class that inherits from \texttt{torch.nn.Module}.}

% Implementation summary:
A trainable $(V, d_{\mathrm{model}})$ table
is initialized from $\mathcal{N}(0,1)$ truncated to $[-3,3]$. Lookup adds an
embedding dimension to the input shape:
\[
E\in\mathbb{R}^{V\times d_{\mathrm{model}}},\qquad
Y_{\ldots,:}=E_{\mathrm{token\_ids}_{\ldots},:}.
\]
\par\noindent
\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={Embedding lookup}]
def forward(self, token_ids: Tensor) -> Tensor:
    return self.weight[token_ids]
\end{lstlisting}
\end{minipage}\par

\problem{rmsnorm}{Root Mean Square Layer Normalization}{1}

\deliverable{Implement \texttt{RMSNorm} as a \texttt{torch.nn.Module}.}

% Implementation summary:
I compute $\operatorname{RMSNorm}(x)=g\odot x/\sqrt{\operatorname{mean}(x^2)+\epsilon}$ over the final dimension, with $g$ initialized to ones. The reduction uses float32 for numerical stability, and the result is cast back to the input dtype.
\par\noindent
\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={RMSNorm forward pass}]
in_dtype = x.dtype
x = x.to(torch.float32)
rms = torch.sqrt(reduce(x**2, "... d_model -> ... 1", "mean") + self.eps)
return (x / rms * self.weight).to(in_dtype)
\end{lstlisting}
\end{minipage}\par


\problem{positionwise\_feedforward}{Implement the position-wise feed-forward network}{2}

\deliverable{Implement the SwiGLU feed-forward network, composed of a SiLU activation and a gated linear unit.}

% Implementation summary:
Three bias-free projections implement $W_2(\operatorname{SiLU}(W_1x)\odot W_3x)$ independently at each token, where $\operatorname{SiLU}(x)=x\sigma(x)$. My default hidden width rounds $8d_{\mathrm{model}}/3$ to the nearest multiple of 64; an explicit \texttt{d\_ff} overrides it.
\par\noindent
\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={SwiGLU computation}]
def swiglu(x: Tensor, w1: Linear, w2: Linear, w3: Linear) -> Tensor:
    return w2(silu(w1(x)) * w3(x))
\end{lstlisting}
\end{minipage}\par


\problem{rope}{Implement RoPE}{2}

\deliverable{Implement a class \texttt{RotaryPositionalEmbedding} that applies RoPE to the input tensor.}

% Implementation summary:
For position $p$ and adjacent pair index
$i=0,\ldots,d_k/2-1$, RoPE applies the rotation
\[
\phi_{p,i}=p\theta^{-2i/d_k},\qquad
\begin{pmatrix}a'\\b'\end{pmatrix}
=\begin{pmatrix}
\cos\phi_{p,i}&-\sin\phi_{p,i}\\
\sin\phi_{p,i}&\cos\phi_{p,i}
\end{pmatrix}
\begin{pmatrix}a\\b\end{pmatrix}.
\]
Sine and cosine tables are cached as nonpersistent buffers and indexed by
the supplied token positions. Rotation uses float32 arithmetic before
casting back to the input dtype.


\problem{softmax}{Implement softmax}{1}

\deliverable{Write a function to apply the softmax operation on a tensor.}

% Implementation summary:
I move the requested dimension to the last axis, subtract its maximum, exponentiate, and divide by the sum before restoring the original axis order. Subtracting the maximum avoids exponential overflow; a slice containing only $-\infty$ still produces NaNs.
\par\noindent
\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={Softmax computation}]
def softmax(x: Tensor, dim: int) -> Tensor:
    moved = x.movedim(dim, -1)
    shifted = moved - reduce(moved, "... feature -> ... 1", "max")
    exp_x = torch.exp(shifted)
    result = exp_x / reduce(exp_x, "... feature -> ... 1", "sum")
    return result.movedim(-1, dim)
\end{lstlisting}
\end{minipage}\par

\problem{scaled\_dot\_product\_attention}{Implement scaled dot-product attention}{5}

\deliverable{Implement the scaled dot-product attention function.}

% Implementation summary:
Attention normalizes scaled dot products
over keys, then forms a weighted sum of values:
\[
S=\frac{QK^\top}{\sqrt{d_k}},\qquad
\widetilde S_{ij}=\begin{cases}
S_{ij},&M_{ij}=\mathrm{True},\\
-\infty,&M_{ij}=\mathrm{False},
\end{cases}\qquad
Y=\operatorname{softmax}(\widetilde S)V.
\]
Einstein summations preserve arbitrary leading batch dimensions. The mask
broadcasts to the score shape; each query must retain at least one allowed
finite score.


\problem{multihead\_self\_attention}{Implement causal multi-head self-attention}{5}

\deliverable{Implement causal multi-head self-attention as a \texttt{torch.nn.Module}.}

% Implementation summary:
Separate projections produce $Q,K,V$;
each is split into $h$ heads of width $d_k=d_{\mathrm{model}}/h$:
\[
(\ldots,T,d_{\mathrm{model}})
\longrightarrow(\ldots,h,T,d_k).
\]
RoPE rotates $Q$ and $K$, using positions $0,\ldots,T-1$ when none are
provided. A lower-triangular mask enforces causality:
\[
M_{ij}=[j\leq i],\qquad
H_r=\operatorname{Attention}(Q_r,K_r,V_r;M),\qquad
Y=\operatorname{Concat}(H_1,\ldots,H_h)W_O^\top.
\]


\problem{transformer\_block}{Implement the Transformer block}{3}

\deliverable{Transformer block code that passes the provided tests.}

% Implementation summary:
Each block uses pre-normalization and two residual connections:
\begin{align*}
y&=x+\operatorname{Attention}(\operatorname{RMSNorm}_1(x)),\\
z&=y+\operatorname{FFN}(\operatorname{RMSNorm}_2(y)).
\end{align*}
Both sublayers preserve the input shape.
\par\noindent
\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={Pre-norm Transformer block}]
def forward(self, x: Tensor) -> Tensor:
    y = x + self.attn(self.ln1(x))
    return y + self.ffn(self.ln2(y))
\end{lstlisting}
\end{minipage}\par

\problem{transformer\_lm}{Implementing the Transformer LM}{3}

\deliverable{A Transformer LM module that passes the provided tests.}

% Implementation summary:
The forward pass maps token IDs to
unnormalized vocabulary logits:
\begin{align*}
H_0&=\operatorname{Embedding}(X),\\
H_\ell&=\operatorname{Block}_\ell(H_{\ell-1}),\quad \ell=1,\ldots,L,\\
Z&=\operatorname{RMSNorm}_{\mathrm{final}}(H_L)W_{\mathrm{out}}^\top
\in\mathbb{R}^{B\times T\times V}.
\end{align*}
Blocks have independent trainable parameters and share one fixed RoPE
cache. Input embeddings and the output projection use separate weights;
softmax is deferred to the loss or sampling routine.


\problem{transformer\_accounting}{Transformer LM resource accounting}{5}

Let $V=50{,}257$, $T=1{,}024$, and let $L,d,h,f$ denote the number of layers,
model width, number of heads, and FFN width, respectively. For XL,
$(L,d,h,f)=(48,1600,25,4288)$; FLOPs below count dense forward matrix
multiplies for one sequence, including masked entries, using $2mnp$ FLOPs
for an $(m\times n)(n\times p)$ product.

\Needspace{6\baselineskip}
\textbf{(a)} How many trainable parameters would a GPT-2 XL-sized model have? How much memory is required to load it (single-precision)?

\deliverable{A one-to-two sentence response.}

% Answer:
With untied embedding/output weights, no biases, and two RMSNorms per block,
the parameter count is
\[
P=2Vd+L(4d^2+3df+2d)+d=1{,}640{,}452{,}800.
\]
FP32 parameters require $4P=6{,}561{,}811{,}200$ bytes, approximately
$6.562$ GB ($6.111$ GiB), excluding buffers and activations.

\Needspace{6\baselineskip}
\textbf{(b)} Identify the matrix multiplies required for a forward pass. How many FLOPs do they require in total?

\deliverable{A list of matrix multiplies (with descriptions), and the total number of FLOPs.}

% Answer:
The matrix multiplies and their costs are listed below;
per-layer attention costs are summed over all heads, where $d_k=d/h$.
\begin{center}
\small
\begin{tabular}{lll}
\toprule
Component & Matrix products & FLOPs per layer \\
\midrule
Q, K, V projections & $3\times(T,d)(d,d)$ & $6Td^2$ \\
Attention scores & $h\times(T,d_k)(d_k,T)$ & $2T^2d$ \\
Attention-weighted values & $h\times(T,T)(T,d_k)$ & $2T^2d$ \\
Output projection & $(T,d)(d,d)$ & $2Td^2$ \\
SwiGLU & $2\times(T,d)(d,f)+(T,f)(f,d)$ & $6Tdf$ \\
\midrule
LM head (once per model) & $(T,d)(d,V)$ & $2TdV$ \\
\bottomrule
\end{tabular}
\end{center}
Thus the total is
\[
F=L(8Td^2+4T^2d+6Tdf)+2TdV
=3{,}516{,}769{,}894{,}400\approx3.517\text{ TFLOPs}.
\]
Embedding lookup and elementwise operations are excluded.

\Needspace{6\baselineskip}
\textbf{(c)} Which parts of the model require the most FLOPs?

\deliverable{A one-to-two sentence response.}

% Answer:
SwiGLU dominates at $57.53\%$ of total FLOPs, followed by
QKV and output projections together at $28.62\%$. The two attention
matrix multiplies account for $9.16\%$, and the LM head accounts for $4.68\%$.

\Needspace{6\baselineskip}
\textbf{(d)} Repeat the analysis with GPT-2 small, medium, and large. How do proportional FLOPs change with model size?

\deliverable{For each model, provide a breakdown of model components and associated FLOPs (as a proportion of total). Include a one-to-two sentence description.}

% Answer:
Using $f$ equal to $8d/3$ rounded to the nearest multiple of 64,
the component shares are:
\begin{center}
\small
\begin{tabular}{lrrrrrr}
\toprule
Model ($f$) & TFLOPs & QKV & Output & Attention & FFN & LM head \\
\midrule
Small (2048) & 0.292 & $14.91\%$ & $4.97\%$ & $13.25\%$ & $39.76\%$ & $27.10\%$ \\
Medium (2752) & 0.830 & $18.62\%$ & $6.21\%$ & $12.42\%$ & $50.05\%$ & $12.70\%$ \\
Large (3392) & 1.769 & $20.49\%$ & $6.83\%$ & $10.93\%$ & $54.30\%$ & $7.45\%$ \\
XL (4288) & 3.517 & $21.47\%$ & $7.16\%$ & $9.16\%$ & $57.53\%$ & $4.68\%$ \\
\bottomrule
\end{tabular}
\end{center}
Here ``Attention'' combines $QK^\top$ and the attention-weighted values.
At fixed $T,V$ and $f\approx8d/3$, projections and FFN scale as $Ld^2$,
attention as $Ld$, and the LM head as $d$; increasing model depth and width
therefore raises the projection/FFN shares and lowers the attention/LM-head shares.

\Needspace{6\baselineskip}
\textbf{(e)} Take GPT-2 XL and increase the context length to 16{,}384. How does the total FLOPs and relative contribution change?

\deliverable{A one-to-two sentence response.}

% Answer:
Increasing $T$ by $16\times$ multiplies projection, FFN,
and LM-head FLOPs by 16, but attention FLOPs by 256:
\[
F_{\mathrm{new}}
=16(F_{\mathrm{old}}-F_{\mathrm{attn}})+256F_{\mathrm{attn}}
\approx133.578\text{ TFLOPs}=37.983\,F_{\mathrm{old}},
\]
where $F_{\mathrm{attn}}=4LT^2d$ at the original context length.
Attention becomes dominant at $61.73\%$ (up from $9.16\%$), while QKV,
output projection, FFN, and LM-head shares fall to $9.04\%$, $3.01\%$,
$24.24\%$, and $1.97\%$, respectively.
